        function getCurrencySymbol() {
            return CURRENCIES[state.currency] || '₽';
        }

        function formatMoney(amount) {
            const sym = getCurrencySymbol();
            const formatted = Math.round(Number(amount) || 0).toLocaleString('ru-RU');
            return ['USD', 'EUR', 'GBP'].includes(state.currency) ? `${sym}${formatted}` : `${formatted} ${sym}`;
        }

        function escapeHtml(str) {
            return String(str || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;').replace(/"/g, '&quot;');
        }

        function getTotalDebtAmount() {
            if (state.debtMode === 'manual') {
                return Number(state.manualDebtAmount) || 0;
            }
            return state.debts.reduce((sum, d) => sum + (Number(d.amount) || 0), 0);
        }

        function getAvailableDebtBudget() {
            const inc = Number(state.budget.income) || 0;
            const rent = Number(state.budget.rent) || 0;
            const food = Number(state.budget.food) || 0;
            const util = Number(state.budget.utility) || 0;
            const savings = Number(state.budget.lifeSavings) || 0;
            return Math.max(0, inc - (rent + food + util + savings));
        }

        function getTotalMinPayments() {
            return state.debts.reduce((sum, d) => sum + (Number(d.minPayment) || 0), 0);
        }

        function calculatePayoffSchedule() {
            const totalDebt = getTotalDebtAmount();
            const monthlyBudget = getAvailableDebtBudget();
            
            if (totalDebt <= 0 || monthlyBudget <= 0) {
                return { totalMonths: 0, totalOverpayment: 0, monthlyData: [] };
            }

            if (state.debts.length === 0) {
                const months = Math.ceil(totalDebt / monthlyBudget);
                const monthlyData = [];
                let rem = totalDebt;
                for (let m = 1; m <= Math.min(months, 120); m++) {
                    rem = Math.max(0, rem - monthlyBudget);
                    monthlyData.push({ month: m, remainingBalance: rem, monthlyInterest: 0 });
                }
                return { totalMonths: months, totalOverpayment: 0, monthlyData };
            }

            let debtsCopy = state.debts.map(d => ({
                id: d.id,
                name: d.name,
                amount: Number(d.amount) || 0,
                rate: Number(d.rate) || 0,
                minPayment: Number(d.minPayment) || 0
            }));

            if (state.debtMode === 'manual' && totalDebt > 0) {
                const autoTotal = state.debts.reduce((s, d) => s + (Number(d.amount) || 0), 0);
                if (autoTotal > 0) {
                    const ratio = totalDebt / autoTotal;
                    debtsCopy.forEach(d => { d.amount = Math.round(d.amount * ratio); });
                }
            }

            let month = 0;
            let totalOverpayment = 0;
            const monthlyData = [];
            const maxMonths = 360;

            while (debtsCopy.some(d => d.amount > 0) && month < maxMonths) {
                month++;
                let extraCash = monthlyBudget;
                let monthInterestSum = 0;

                debtsCopy.forEach(d => {
                    if (d.amount > 0) {
                        const interest = (d.amount * (d.rate / 100)) / 12;
                        monthInterestSum += interest;
                        d.amount += interest;
                        totalOverpayment += interest;
                    }
                });

                debtsCopy.forEach(d => {
                    if (d.amount > 0) {
                        const pay = Math.min(d.amount, d.minPayment, extraCash);
                        d.amount -= pay;
                        extraCash -= pay;
                    }
                });

                const activeDebts = debtsCopy.filter(d => d.amount > 0);
                if (state.strategy === 'snowball') {
                    activeDebts.sort((a, b) => a.amount - b.amount);
                } else {
                    activeDebts.sort((a, b) => b.rate - a.rate);
                }

                for (let d of activeDebts) {
                    if (extraCash <= 0) break;
                    const extraPay = Math.min(d.amount, extraCash);
                    d.amount -= extraPay;
                    extraCash -= extraPay;
                }

                const currentRemaining = debtsCopy.reduce((sum, d) => sum + d.amount, 0);
                monthlyData.push({
                    month,
                    remainingBalance: Math.round(currentRemaining),
                    monthlyInterest: Math.round(monthInterestSum)
                });
            }

            return {
                totalMonths: month,
                totalOverpayment: Math.round(totalOverpayment),
                monthlyData
            };
        }

        function renderPayoffChart(monthlyData) {
            if (!window.Chart) return;
            const ctx = document.getElementById('payoffChart')?.getContext('2d');
            if (!ctx) return;

            if (payoffChart) payoffChart.destroy();

            const labels = monthlyData.map(d => `M${d.month}`);
            const data = monthlyData.map(d => d.remainingBalance);

            payoffChart = new Chart(ctx, {
                type: 'line',
                data: {
                    labels,
                    datasets: [{
                        label: 'Остаток долга',
                        data,
                        borderColor: '#ef4444',
                        backgroundColor: 'rgba(239, 68, 68, 0.15)',
                        fill: true,
                        tension: 0.3,
                        pointRadius: 2,
                        pointBackgroundColor: '#ef4444'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: { legend: { display: false } },
                    scales: {
                        x: { ticks: { color: '#64748b', font: { size: 9 } }, grid: { color: 'rgba(239, 68, 68, 0.05)' } },
                        y: { ticks: { color: '#64748b', font: { size: 9 } }, grid: { color: 'rgba(239, 68, 68, 0.08)' } }
                    }
                }
            });
        }

        function switchTab(tabName) {
            triggerHaptic('impact', 'light');
            ['budget', 'debts', 'plan'].forEach(name => {
                const section = document.getElementById(`section-${name}`);
                const btn = document.getElementById(`tab-${name}`);
                if (name === tabName) {
                    section.classList.remove('hidden');
                    btn.className = "tab-btn text-xs font-bold py-2.5 rounded-lg transition-all text-center flex items-center justify-center gap-1.5 scarlet-btn text-white";
                } else {
                    section.classList.add('hidden');
                    btn.className = "tab-btn text-xs font-bold py-2.5 rounded-lg transition-all text-center flex items-center justify-center gap-1.5 text-slate-400 hover:text-white";
                }
            });
            if (tabName === 'plan') updatePlanView();
        }

        function saveData() {
            try {
                localStorage.setItem('scarlet_spider_debt_data', JSON.stringify({
                    currency: state.currency,
                    debtMode: state.debtMode,
                    manualDebtAmount: state.manualDebtAmount,
                    budget: state.budget,
                    debts: state.debts,
                    strategy: state.strategy
                }));
            } catch(e) {}
        }

        function loadData() {
            const savedData = localStorage.getItem('scarlet_spider_debt_data');
            if (savedData) {
                try {
                    const parsed = JSON.parse(savedData);
                    if (parsed.currency) state.currency = parsed.currency;
                    if (parsed.budget) state.budget = parsed.budget;
                    if (parsed.debts) state.debts = parsed.debts;
                    if (parsed.strategy) state.strategy = parsed.strategy;
                    if (parsed.debtMode) state.debtMode = parsed.debtMode;
                    if (parsed.manualDebtAmount !== undefined) state.manualDebtAmount = parsed.manualDebtAmount;
                } catch (e) {}
            }
        }

        function updateDebtProgressBar() {
            const currentTotal = getTotalDebtAmount();
            document.getElementById('total-debt-sum').innerText = formatMoney(currentTotal);
            
            const maxRef = Math.max(1000000, currentTotal);
            const percentLeft = currentTotal === 0 ? 0 : Math.min(100, Math.round((currentTotal / maxRef) * 100));
            document.getElementById('debt-progress-percent').innerText = `${currentTotal === 0 ? '0%' : percentLeft + '%'}`;
            document.getElementById('debt-progress-bar').style.width = `${currentTotal === 0 ? '0%' : Math.max(5, percentLeft) + '%'}`;
        }

        function updateBudgetSummary() {
            const available = getAvailableDebtBudget();
            const minPayments = getTotalMinPayments();

            document.getElementById('debt-budget-val').innerText = formatMoney(available);
            document.getElementById('min-payments-total-val').innerText = formatMoney(minPayments);

            const statusBox = document.getElementById('budget-status-box');
            const statusIcon = document.getElementById('budget-status-icon');
            const statusText = document.getElementById('budget-status-text');

            if (available < minPayments) {
                statusBox.className = "p-3 rounded-xl text-xs flex items-center gap-2.5 bg-red-950/60 border border-red-600/50 text-red-200";
                statusIcon.setAttribute('data-lucide', 'alert-octagon');
                statusText.innerText = `Внимание! Доступно (${formatMoney(available)}) МЕНЬШЕ минимальных обязательств (${formatMoney(minPayments)})!`;
            } else if (available === minPayments) {
                statusBox.className = "p-3 rounded-xl text-xs flex items-center gap-2.5 bg-amber-950/50 border border-amber-600/40 text-amber-200";
                statusIcon.setAttribute('data-lucide', 'alert-triangle');
                statusText.innerText = `Бюджета хватает только на минимальные платежи.`;

        }

        function updatePlanView() {
            const planSection = document.getElementById('section-plan');
            if (planSection && planSection.classList.contains('hidden')) return;

            const res = calculatePayoffSchedule();

            const badge = document.getElementById('plan-strategy-badge');
            if (badge) {
                badge.innerText = state.strategy === 'snowball' ? 'Снежный ком' : 'Лавина';
            }

            const monthsEl = document.getElementById('result-months');
            const overpayEl = document.getElementById('result-overpayment');
            if (monthsEl) monthsEl.innerText = `${res.totalMonths} мес.`;
            if (overpayEl) overpayEl.innerText = formatMoney(res.totalOverpayment);

            const scheduleContainer = document.getElementById('monthly-schedule');
            if (scheduleContainer) {
                if (res.monthlyData.length === 0) {
                    scheduleContainer.innerHTML = `
                        <div class="text-center py-4 text-slate-500 text-xs">
                            Заполните параметры дохода и обязательств для расчета.
                        </div>
                    `;
                } else {
                    scheduleContainer.innerHTML = res.monthlyData.map(item => `
                        <div class="flex justify-between items-center bg-slate-950/60 p-2.5 rounded-xl border border-red-900/20 text-xs">
                            <span class="font-bold text-slate-300">Месяц ${item.month}</span>
                            <div class="text-right">
                                <span class="font-bold text-white block">${formatMoney(item.remainingBalance)} остаток</span>
                                <span class="text-[10px] text-red-400 font-medium">+${formatMoney(item.monthlyInterest)} % за месяц</span>
                            </div>
                        </div>
                    `).join('');
                }
            }

            renderPayoffChart(res.monthlyData);
        }
